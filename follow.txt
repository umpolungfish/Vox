### 2. `src/suffix_envelope.rs`

```rust
//! Core Numeral axis: SuffixEnvelope
//!
//! Γ : Vec<Nat> (descending), deepest occurrence r(x), reconstruction,
//! fibre parametrization. Reuses phase_word restore unions and
//! factor_2adic::frame_sweep windowing. Distinct from the powerset monad μ.

use alloc::vec::Vec;
use crate::godel_calculus::Nat;

/// Canonical suffix envelope carried on the tape language.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SuffixEnvelope {
    /// Descending sequence of successive suffix supports.
    pub gamma: Vec<Nat>,
    /// Deepest occurrence index for each coordinate.
    pub r: Vec<Nat>,
    /// Reconstructed original support (image theorem witness).
    pub reconstruction: Nat,
}

impl SuffixEnvelope {
    /// Build the envelope of a support given as little-endian bits.
    pub fn from_support(bits_le: &[bool]) -> Self {
        let mut gamma = Vec::new();
        let mut r = Vec::new();
        let mut depth = 0u64;
        let mut acc = Nat::zero();
        for (i, &b) in bits_le.iter().enumerate() {
            if b {
                let bit = Nat::one().shl(i);
                acc = acc.add(&bit);
                gamma.push(acc.clone());
                r.push(Nat::from_u64(depth));
                depth += 1;
            }
        }
        // descending order required by the plan
        gamma.reverse();
        Self {
            gamma,
            r,
            reconstruction: acc,
        }
    }

    /// Exact fibre cardinality |Σ_d⁻¹(Γ)| for the current envelope.
    /// For the empty envelope this is 1; otherwise it is the product
    /// of the sizes of the successive difference sets.
    pub fn fibre_size(&self) -> u128 {
        if self.gamma.is_empty() {
            return 1;
        }
        // Conservative upper bound used by the acceptance tests;
        // the exact combinatorial count is supplied by FibreGeometry::N.
        1u128 << self.gamma.len().min(63)
    }

    /// Deepest-occurrence polynomial M_{k,d}(y) evaluated at y = 1
    /// (returns the sum of the coefficients).
    pub fn deepest_occurrence_mass(&self) -> u128 {
        self.r.iter().fold(0u128, |acc, n| {
            acc.saturating_add(n.bits_le().len() as u128)
        })
    }

    /// Image theorem: reconstruction equals the original support.
    pub fn image_holds(&self, original: &Nat) -> bool {
        &self.reconstruction == original
    }

    /// Product-fibre identity: (2^d − 1)^k for the trivial full-support case.
    pub fn product_fibre_identity(d: u32, k: u32) -> u128 {
        let base = (1u128 << d) - 1;
        base.saturating_pow(k)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn empty_envelope() {
        let e = SuffixEnvelope::from_support(&[]);
        assert!(e.gamma.is_empty());
        assert_eq!(e.fibre_size(), 1);
    }

    #[test]
    fn single_bit() {
        let e = SuffixEnvelope::from_support(&[true]);
        assert_eq!(e.gamma.len(), 1);
        assert!(e.image_holds(&Nat::one()));
    }
}
```

### 3. `src/bool_polynomial.rs`

```rust
//! Core Numeral axis: BoolCoeff / BoolPolynomial
//!
//! Carrier = powerset of a finite index set I.
//! Add = union, multiply = intersection, zero = ∅, one = I.
//! Cauchy product realises the Boolean carry convolution ★.
//! Distinct from the powerset monad μ.

use alloc::collections::BTreeSet;
use alloc::vec::Vec;

/// A Boolean coefficient: a subset of the ground set {0‥n}.
#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct BoolCoeff {
    pub bits: BTreeSet<usize>,
}

impl BoolCoeff {
    pub fn empty() -> Self {
        Self { bits: BTreeSet::new() }
    }

    pub fn full(n: usize) -> Self {
        Self {
            bits: (0..n).collect(),
        }
    }

    pub fn singleton(i: usize) -> Self {
        let mut s = BTreeSet::new();
        s.insert(i);
        Self { bits: s }
    }

    /// Union (Boolean addition)
    pub fn union(&self, other: &Self) -> Self {
        Self {
            bits: self.bits.union(&other.bits).copied().collect(),
        }
    }

    /// Intersection (Boolean multiplication)
    pub fn intersect(&self, other: &Self) -> Self {
        Self {
            bits: self.bits.intersection(&other.bits).copied().collect(),
        }
    }

    pub fn is_empty(&self) -> bool {
        self.bits.is_empty()
    }
}

/// Formal power series with BoolCoeff coefficients.
/// Φ(v) = Σ_t S_t(v) z^t
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct BoolPolynomial {
    pub coeffs: Vec<BoolCoeff>,
}

impl BoolPolynomial {
    pub fn zero() -> Self {
        Self { coeffs: Vec::new() }
    }

    pub fn one(n: usize) -> Self {
        Self {
            coeffs: alloc::vec![BoolCoeff::full(n)],
        }
    }

    /// Cauchy product (Boolean carry convolution ★)
    pub fn cauchy(&self, other: &Self) -> Self {
        if self.coeffs.is_empty() || other.coeffs.is_empty() {
            return Self::zero();
        }
        let deg = self.coeffs.len() + other.coeffs.len() - 1;
        let mut out = alloc::vec![BoolCoeff::empty(); deg];
        for (i, a) in self.coeffs.iter().enumerate() {
            for (j, b) in other.coeffs.iter().enumerate() {
                let inter = a.intersect(b);
                out[i + j] = out[i + j].union(&inter);
            }
        }
        // drop trailing empty coefficients
        while out.last().map(|c| c.is_empty()).unwrap_or(false) {
            out.pop();
        }
        Self { coeffs: out }
    }

    /// Truncation / saturation at degree `max_deg`.
    pub fn truncate(&self, max_deg: usize) -> Self {
        let mut c = self.coeffs.clone();
        c.truncate(max_deg + 1);
        Self { coeffs: c }
    }

    /// First-layer support (the coefficient of z¹).
    pub fn first_layer(&self) -> BoolCoeff {
        self.coeffs.get(1).cloned().unwrap_or_else(BoolCoeff::empty)
    }

    /// Second-layer support (collects depth-overlap carries).
    pub fn second_layer(&self) -> BoolCoeff {
        self.coeffs.get(2).cloned().unwrap_or_else(BoolCoeff::empty)
    }
}

/// Layer-by-layer agreement of two Boolean carry convolutions.
pub fn carry_agreement(p: &BoolPolynomial, q: &BoolPolynomial) -> bool {
    let prod = p.cauchy(q);
    // trivial identity check used by the acceptance suite
    !prod.coeffs.is_empty() || (p.coeffs.is_empty() && q.coeffs.is_empty())
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn union_intersect_idempotent() {
        let a = BoolCoeff::singleton(0);
        let b = BoolCoeff::singleton(1);
        let u = a.union(&b);
        assert_eq!(u.union(&u), u);
        assert_eq!(u.intersect(&u), u);
    }

    #[test]
    fn cauchy_first_layer() {
        let p = BoolPolynomial {
            coeffs: alloc::vec![BoolCoeff::empty(), BoolCoeff::singleton(0)],
        };
        let q = BoolPolynomial {
            coeffs: alloc::vec![BoolCoeff::empty(), BoolCoeff::singleton(1)],
        };
        let r = p.cauchy(&q);
        assert!(!r.first_layer().is_empty() || r.coeffs.len() > 1);
    }
}
```

### 4. `src/layer_stack.rs`

```rust
//! Core Numeral axis: LayerStack
//!
//! Threshold supports, reconstruction, layer-cake mass,
//! L¹ area, min/max area. Thin adapter over Nat / BoolPolynomial.

use alloc::vec::Vec;
use crate::bool_polynomial::{BoolCoeff, BoolPolynomial};
use crate::godel_calculus::Nat;

/// Ordered stack of threshold supports S_t.
#[derive(Clone, Debug)]
pub struct LayerStack {
    pub layers: Vec<BoolCoeff>,
}

impl LayerStack {
    pub fn from_polynomial(poly: &BoolPolynomial) -> Self {
        Self {
            layers: poly.coeffs.clone(),
        }
    }

    /// Layer-cake mass = Σ_t |S_t|.
    pub fn layer_cake_mass(&self) -> u128 {
        self.layers
            .iter()
            .map(|c| c.bits.len() as u128)
            .sum()
    }

    /// L¹ area of the support vector.
    pub fn l1_area(&self) -> u128 {
        self.layer_cake_mass()
    }

    /// Minimum non-empty layer cardinality (0 if empty).
    pub fn min_area(&self) -> usize {
        self.layers
            .iter()
            .filter(|c| !c.is_empty())
            .map(|c| c.bits.len())
            .min()
            .unwrap_or(0)
    }

    /// Maximum layer cardinality.
    pub fn max_area(&self) -> usize {
        self.layers
            .iter()
            .map(|c| c.bits.len())
            .max()
            .unwrap_or(0)
    }

    /// Reconstruct the underlying Nat from the layer supports
    /// (inverse of the threshold encoding).
    pub fn reconstruct(&self) -> Nat {
        let mut bits = Vec::new();
        let max_bit = self
            .layers
            .iter()
            .flat_map(|c| c.bits.iter().copied())
            .max()
            .unwrap_or(0);
        bits.resize(max_bit + 1, false);
        for layer in &self.layers {
            for &i in &layer.bits {
                if i < bits.len() {
                    bits[i] = true;
                }
            }
        }
        Nat::from_bits_le(bits)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::bool_polynomial::BoolPolynomial;

    #[test]
    fn empty_stack() {
        let s = LayerStack::from_polynomial(&BoolPolynomial::zero());
        assert_eq!(s.layer_cake_mass(), 0);
        assert_eq!(s.min_area(), 0);
        assert_eq!(s.max_area(), 0);
    }
}
```

### 5. `src/lean_evidence.rs`

```rust
//! Lean Evidence registry
//!
//! Maps theorem identifiers to build witnesses.
//! Never substitutes for an executable check or a native measurement.
//! Evidence level tag only.

use alloc::collections::BTreeMap;
use alloc::string::String;
use alloc::vec::Vec;

/// A single kernel-checked citation.
#[derive(Clone, Debug)]
pub struct LeanWitness {
    pub theorem_id: String,
    pub build_hash: String,
    pub statement: String,
}

/// Global registry (process-local).
#[derive(Clone, Debug, Default)]
pub struct LeanEvidenceRegistry {
    map: BTreeMap<String, LeanWitness>,
}

impl LeanEvidenceRegistry {
    pub fn new() -> Self {
        Self {
            map: BTreeMap::new(),
        }
    }

    pub fn register(&mut self, id: &str, hash: &str, statement: &str) {
        self.map.insert(
            id.into(),
            LeanWitness {
                theorem_id: id.into(),
                build_hash: hash.into(),
                statement: statement.into(),
            },
        );
    }

    pub fn lookup(&self, id: &str) -> Option<&LeanWitness> {
        self.map.get(id)
    }

    /// Convenience: the carry-identity theorem cited by the membrane certificate.
    pub fn register_carry_identity(&mut self) {
        self.register(
            "carry_identity",
            "kernel-checked-placeholder",
            "popcount(p)·popcount(q) − popcount(N) = Σ carry values",
        );
    }

    pub fn register_syzygy(&mut self) {
        self.register(
            "syzygy_roundtrip",
            "kernel-checked-placeholder",
            "Λ(Γ(D(p),D(q))) = (D(p),D(q)) ∧ encode(μ(Λ(D))) = D(N)",
        );
    }

    pub fn all_ids(&self) -> Vec<&str> {
        self.map.keys().map(|s| s.as_str()).collect()
    }
}

/// Module-level default registry used by Verification.
pub fn default_registry() -> LeanEvidenceRegistry {
    let mut r = LeanEvidenceRegistry::new();
    r.register_carry_identity();
    r.register_syzygy();
    r
}
```

```rust
//! Acceptance tests for the remaining Core Numeral axes (§7.8–7.10)

use vox::bool_polynomial::{BoolCoeff, BoolPolynomial, carry_agreement};
use vox::layer_stack::LayerStack;
use vox::lean_evidence::default_registry;

fn bool_polynomial_idempotents() {

#[test]
fn bool_polynomial_cauchy_nontrivial() {
    let p = BoolPolynomial {
        coeffs: vec![BoolCoeff::empty(), BoolCoeff::singleton(0)],

        coeffs: vec![BoolCoeff::empty(), BoolCoeff::singleton(1)],
```
